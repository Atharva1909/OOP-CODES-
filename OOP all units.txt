using namespace std;
class Fraction
{
private:
 int numerator;
 int denominator;
 int gcd(int a, int b)
 {
 while(b != 0)
 {
 int temp = b;
 b = a % b;
 a = temp;
 }
 return a;
 }
public:
 void input()
 {
 cout << "Enter numerator: ";
 cin >> numerator;
 cout << "Enter denominator: ";
 cin >> denominator;
 if(denominator == 0)
 {
 cout << "Denominator cannot be zero.\n";
 denominator = 1;
 }
 }
 void simplify()
 {
 int g = gcd(numerator, denominator);
 numerator = numerator / g;
 denominator = denominator / g;
 if(denominator < 0)
 {
 numerator = -numerator;
 denominator = -denominator;
 }
 }
 Fraction operator+(Fraction f)
 {
 Fraction result;
 result.numerator =
 numerator * f.denominator +
 f.numerator * denominator;
 result.denominator =
 denominator * f.denominator;
 result.simplify();
 return result;
 }
 Fraction operator-(Fraction f)
 {
 Fraction result;
 result.numerator =
 numerator * f.denominator -
 f.numerator * denominator;
 result.denominator =
 denominator * f.denominator;
 result.simplify();
 return result;
 }
 Fraction operator*(Fraction f)
 {
 Fraction result;
 result.numerator =
 numerator * f.numerator;

 return result;

 return result;
