## Q6. Student Class – Total, Percentage and Grade

```cpp
#include <iostream>
using namespace std;

class Student {
    int marks[5], total;
    float percent;

public:
    void input() {
        for(int i=0;i<5;i++) cin >> marks[i];
    }

    void calculate() {
        total=0;
        for(int i=0;i<5;i++) total+=marks[i];
        percent=total/5.0;
    }

    void display() {
        char grade;
        if(percent>=90) grade='A';
        else if(percent>=75) grade='B';
        else if(percent>=60) grade='C';
        else if(percent>=50) grade='D';
        else grade='F';

        cout<<"Total = "<<total<<"\n";
        cout<<"Percentage = "<<percent<<"\n";
        cout<<"Grade = "<<grade;
    }
};

int main() {
    Student s;
    s.input();
    s.calculate();
    s.display();
}
```

## Q46. Complex Number Arithmetic using Operator Overloading

```cpp
#include <iostream>
using namespace std;

class Complex {
    int r,i;

public:
    Complex(int a=0,int b=0) {
        r=a;
        i=b;
    }

    Complex operator+(Complex c) {
        return Complex(r+c.r,i+c.i);
    }

    Complex operator-(Complex c) {
        return Complex(r-c.r,i-c.i);
    }

    Complex operator*(Complex c) {
        return Complex(r*c.r-i*c.i,r*c.i+i*c.r);
    }

    Complex operator/(Complex c) {
        int d=c.r*c.r+c.i*c.i;
        return Complex((r*c.r+i*c.i)/d,
                       (i*c.r-r*c.i)/d);
    }

    void show() {
        cout<<r<<" + "<<i<<"i\n";
    }
};

int main() {
    Complex a(8,4),b(2,2);

    (a+b).show();
    (a-b).show();
    (a*b).show();
    (a/b).show();
}
```

## Q49. Implicit and Explicit Conversion Between User-Defined Classes

```cpp
#include <iostream>
using namespace std;

class Meter;

class Feet {
    float f;

public:
    Feet(float x) {
        f=x;
    }

    float get() {
        return f;
    }

    operator Meter();
};

class Meter {
    float m;

public:
    Meter(float x=0) {
        m=x;
    }

    void show() {
        cout<<m<<" meters";
    }

    operator Feet() {
        return Feet(m*3.28084);
    }
};

Feet::operator Meter() {
    return Meter(f*0.3048);
}

int main() {
    Feet f(10);

    Meter m=f;       // implicit
    m.show();

    Feet x=Meter(5); // explicit assignment style conversion
    cout<<"\n"<<x.get()<<" feet";
}
```

#include <iostream>

## Q60. Complete Runtime Polymorphism Design

#include <iostream>

    virtual void area()=0;

class Circle : public Shape {

        r=x;
    }

        cout<<"Circle = "<<3.14*r*r<<"\n";
    }

    Rectangle(float x,float y) {
        l=x;
        b=y;
    }

        cout<<"Rectangle = "<<l*b<<"\n";
    }

int main() {

    p=new Rectangle(4,5);
